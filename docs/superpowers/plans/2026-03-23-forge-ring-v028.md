  ```js
  function getMonthList() {
    const ll = effL(), hh = effH();
    const today = getDate();
    const logKeys = Object.keys(ll).sort();
    const earliest = logKeys[0] || today;

    let [ey, em] = earliest.slice(0, 7).split('-').map(Number);
    const [ty, tm] = today.slice(0, 7).split('-').map(Number);
    const months = [];

    while (ey < ty || (ey === ty && em <= tm)) {
      const daysInMonth = new Date(ey, em, 0).getDate();
      const startDow = new Date(ey, em - 1, 1).getDay(); // 0=Sun
      const prefix = `${ey}-${String(em).padStart(2, '0')}`;

      const dayPcts = [];
      for (let d = 1; d <= daysInMonth; d++) {
        const ds = `${prefix}-${String(d).padStart(2, '0')}`;
        const p = dayPct(ds, hh, ll);
        if (p !== null) dayPcts.push(p); // null = no habits, skip
      }

      const hasData = dayPcts.length > 0;
      const avg = hasData
        ? Math.max(0, Math.round(dayPcts.reduce((a, b) => a + b, 0) / dayPcts.length))
        : 0;

      months.push({
        year: ey,
        month: em,
        prefix,
        daysInMonth,
        startDow,
        pct: avg,
        hasData,
        word: scoreWord(avg)
      });

      em++;
      if (em > 12) { em = 1; ey++; }
    }

    return months;
  }
  ```

### Task 4: SVG Ring — `drawRing(mode)`

This function renders the wedge paths and updates the ring center text.

- [ ] **Add `drawRing(mode)` to `index.html`:**

  ```js
  function drawRing(mode) {
    const m = monthList[currentMonthIdx];
    if (!m) return;

    const cx = 128, cy = 128, ro = 122, ri = 82;
    const totalDays = m.daysInMonth;
    const today = getDate();
    const todayMonthPrefix = today.slice(0, 7);
    const isCurrentMonth = m.prefix === todayMonthPrefix;
    const todayDay = isCurrentMonth ? parseInt(today.slice(8), 10) : null;

    const hh = effH();
    const ll = effL();

    function polarToXY(angleDeg, r) {
      const rad = (angleDeg - 90) * (Math.PI / 180);
      return { x: cx + r * Math.cos(rad), y: cy + r * Math.sin(rad) };
    }

    function wedgePath(startDeg, endDeg) {
      const s1 = polarToXY(startDeg, ri);
      const s2 = polarToXY(startDeg, ro);
      const e1 = polarToXY(endDeg, ri);
      const e2 = polarToXY(endDeg, ro);
      const large = (endDeg - startDeg) > 180 ? 1 : 0;
      return [
        `M ${s2.x} ${s2.y}`,
        `A ${ro} ${ro} 0 ${large} 1 ${e2.x} ${e2.y}`,
        `L ${e1.x} ${e1.y}`,
        `A ${ri} ${ri} 0 ${large} 0 ${s1.x} ${s1.y}`,
        'Z'
      ].join(' ');
    }

    function wedgeFill(dayNum, pct) {
      const dateStr = `${m.prefix}-${String(dayNum).padStart(2, '0')}`;
      if (isCurrentMonth && dayNum === todayDay) return '#e8c84c';
      if (isCurrentMonth && dayNum > todayDay) return 'rgba(255,255,255,0.04)';
      if (pct === null) return 'rgba(255,255,255,0.02)';
      if (pct === 100) return 'rgba(201,168,76,0.93)';
      if (pct >= 70) return 'rgba(201,168,76,0.60)';
      if (pct >= 40) return 'rgba(201,168,76,0.32)';
      if (pct >= 1) return 'rgba(200,50,40,0.45)';
      return 'rgba(255,255,255,0.07)';
    }

    const degPerDay = 360 / totalDays;
    const pad = 1.5; // padding angle in degrees between wedges

    // --- Build wedge SVG ---
    function buildWedges() {
      const g = document.getElementById('fr-wedges');
      if (!g) return;
      let html = '';

      for (let d = 1; d <= totalDays; d++) {
        const dateStr = `${m.prefix}-${String(d).padStart(2, '0')}`;
        const pct = dayPct(dateStr, hh, ll);
        const startDeg = (d - 1) * degPerDay + pad / 2;
        const endDeg = d * degPerDay - pad / 2;
        const fill = wedgeFill(d, pct);
        const isToday = isCurrentMonth && d === todayDay;
        const isFuture = isCurrentMonth && d > todayDay;
        const isNull = pct === null;

        let attrs = '';
        let cls = '';
        let label = '';

        if (isToday) {
          cls = 'fr-wedge-today';
          attrs = `role="button" tabindex="0" aria-label="Today, in progress" onclick="tapWedge(${d - 1}, '${dateStr}', ${pct ?? 0})"`;
        } else if (isFuture || isNull) {
          attrs = 'aria-hidden="true"';
        } else {
          const word = scoreWord(pct ?? 0);
          label = `Day ${d}: ${pct ?? 0}% — ${word}`;
          attrs = `role="button" tabindex="0" aria-label="${label}" onclick="tapWedge(${d - 1}, '${dateStr}', ${pct ?? 0})"`;
        }

        const path = wedgePath(startDeg, endDeg);
        const style = mode === 'assemble'
          ? `animation: wedgeIn 0.06s calc(${d - 1} * 14ms) cubic-bezier(0.22,1,0.36,1) both;`
          : '';

        // Failed wedge: add small ✕ centered on wedge midpoint
        let extraEl = '';
        if (!isToday && !isFuture && !isNull && pct !== null && pct >= 1 && pct < 40) {
          const midDeg = (startDeg + endDeg) / 2;
          const midR = (ri + ro) / 2;
          const mp = polarToXY(midDeg, midR);
          extraEl = `<text x="${mp.x}" y="${mp.y}" text-anchor="middle" dominant-baseline="central"
            fill="rgba(255,255,255,0.5)" font-size="7" pointer-events="none" aria-hidden="true">✕</text>`;
        }

        html += `<path d="${path}" fill="${fill}" stroke="none" class="${cls}" style="${style}" ${attrs}></path>${extraEl}`;
      }

      g.innerHTML = html;
    }

    // --- Update ring center ---
    function updateCenter() {
      const pctEl = document.getElementById('fr-rc-pct');
      const wordEl = document.getElementById('fr-rc-word');
      const daysEl = document.getElementById('fr-rc-days');
      if (!pctEl || !wordEl || !daysEl) return;

      const forgedDays = m.hasData
        ? (() => {
            let count = 0;
            for (let d = 1; d <= totalDays; d++) {
              const ds = `${m.prefix}-${String(d).padStart(2, '0')}`;
              const p = dayPct(ds, hh, ll);
              if (p !== null && p >= 50) count++;
            }
            return count;
          })()
        : 0;

      pctEl.textContent = m.hasData ? `${m.pct}%` : '—';
      wordEl.textContent = m.hasData ? m.word : 'No data';
      daysEl.textContent = `${forgedDays} day${forgedDays !== 1 ? 's' : ''} forged`;
    }

    if (mode === 'transition') {
      const svg = document.getElementById('fr-ring-svg');
      if (!svg) return;
      svg.style.transition = 'opacity 150ms ease-in';
      svg.style.opacity = '0';
      setTimeout(() => {
        buildWedges();
        updateCenter();
        svg.style.transition = 'opacity 220ms cubic-bezier(0.22,1,0.36,1)';
        svg.style.opacity = '1';
      }, 160);
    } else {
      buildWedges();
      updateCenter();
      if (mode !== 'assemble') {
        const svg = document.getElementById('fr-ring-svg');
        if (svg) { svg.style.transition = ''; svg.style.opacity = '1'; }
      }
    }
  }
  ```

### Task 5: Ring 3D Tilt + Drag Affordance — `initRingTilt()`

- [ ] **Add `initRingTilt()` to `index.html`:**

  ```js
  function initRingTilt() {
    const tilt = document.getElementById('fr-ring-tilt');
    if (!tilt) return;

    let rotX = 0, rotY = 0;
    let velX = 0, velY = 0;
    let dragging = false;
    let lastX = 0, lastY = 0;

    function applyTransform() {
      tilt.style.transform = `rotateX(${rotX}deg) rotateY(${rotY}deg)`;
    }

    function startDrag(x, y) {
      dragging = true;
      lastX = x; lastY = y;
      velX = 0; velY = 0;
      tilt.style.cursor = 'grabbing';
      tilt.style.transition = '';
    }

    function moveDrag(x, y) {
      if (!dragging) return;
      const dx = x - lastX;
      const dy = y - lastY;
      lastX = x; lastY = y;
      rotY = Math.max(-18, Math.min(18, rotY + dx * 0.4));
      rotX = Math.max(-22, Math.min(22, rotX - dy * 0.4));
      velX = -dy * 0.4;
      velY = dx * 0.4;
      applyTransform();
    }

    function endDrag() {
      if (!dragging) return;
      dragging = false;
      tilt.style.cursor = 'grab';
      startAnimFrame();
    }

    tilt.addEventListener('mousedown', e => startDrag(e.clientX, e.clientY));
    window.addEventListener('mousemove', e => moveDrag(e.clientX, e.clientY));
    window.addEventListener('mouseup', endDrag);

    tilt.addEventListener('touchstart', e => {
      const t = e.touches[0];
      startDrag(t.clientX, t.clientY);
    }, { passive: true });
    tilt.addEventListener('touchmove', e => {
      const t = e.touches[0];
      moveDrag(t.clientX, t.clientY);
    }, { passive: true });
    tilt.addEventListener('touchend', endDrag);

    tilt.style.cursor = 'grab';

    // Store tilt state on module scope for animFrame access
    window._frTilt = { get rotX() { return rotX; }, get rotY() { return rotY; },
      get velX() { return velX; }, get velY() { return velY; },
      set rotX(v) { rotX = v; }, set rotY(v) { rotY = v; },
      set velX(v) { velX = v; }, set velY(v) { velY = v; },
      get dragging() { return dragging; },
      applyTransform, tilt };

    // Drag affordance hint: auto-tilt 8° then spring back, once per session
    if (!sessionStorage.getItem('fr-hint-shown')) {
      sessionStorage.setItem('fr-hint-shown', '1');
      let hintT = 0;
      const hintDuration = 1200;
      const hintStart = performance.now();
      function hintFrame(now) {
        hintT = (now - hintStart) / hintDuration;
        if (hintT < 0.5) {
          rotY = 8 * (hintT / 0.5);
        } else if (hintT < 1) {
          rotY = 8 * (1 - (hintT - 0.5) / 0.5);
        } else {
          rotY = 0;
          applyTransform();
          return;
        }
        applyTransform();
        requestAnimationFrame(hintFrame);
      }
      setTimeout(() => requestAnimationFrame(hintFrame), 600);
    }
  }
  ```

  ```js
  function changeMonth(delta) {
    const newIdx = currentMonthIdx + delta;
    if (newIdx < 0 || newIdx >= monthList.length) return;

    const dir = delta > 0 ? 1 : -1;
    currentMonthIdx = newIdx;
  ```

  ```js
  function initTerrainDrag() {
    const stage = document.getElementById('fr-terrain-stage');
    const inner = document.getElementById('fr-terrain-inner');
    if (!stage || !inner) return;

    let dragging = false;
    let startX = 0;
    let lastDX = 0;

    window._frTerrain = window._frTerrain || {};

    function getOffset() { return window._frTerrain.offset || 0; }
    function setOffset(v) {
      const totalW = window._frTerrain.totalW || inner.scrollWidth;
      const viewportW = stage.offsetWidth || 320;
      const clamped = Math.max(-(totalW - viewportW), Math.min(0, v));
      window._frTerrain.offset = clamped;
      inner.style.transform = `translateX(${clamped}px)`;
    }

    function startTDrag(x) {
      dragging = true;
      startX = x;
      lastDX = 0;
      window._frTerrain.velocity = 0;
      stage.style.cursor = 'grabbing';
    }

    function moveTDrag(x) {
      if (!dragging) return;
      const dx = x - startX;
      startX = x;
      lastDX = dx;
      setOffset(getOffset() + dx);
    }

    function endTDrag() {
      if (!dragging) return;
      dragging = false;
      stage.style.cursor = 'grab';
      window._frTerrain.velocity = lastDX;
      startAnimFrame();
    }

    stage.addEventListener('mousedown', e => startTDrag(e.clientX));
    window.addEventListener('mousemove', e => moveTDrag(e.clientX));
    window.addEventListener('mouseup', endTDrag);

    stage.addEventListener('touchstart', e => {
      startTDrag(e.touches[0].clientX);
    }, { passive: true });
    stage.addEventListener('touchmove', e => {
      moveTDrag(e.touches[0].clientX);
      e.preventDefault();
    }, { passive: false });
    stage.addEventListener('touchend', endTDrag);

    stage.style.cursor = 'grab';

    window._frTerrain.dragging = false;
    Object.defineProperty(window._frTerrain, 'isDragging', {
      get: () => dragging, configurable: true
    });
  }
  ```

### Task 10: Shared rAF Loop — `animFrame()` and `startAnimFrame()`

- [ ] **Add the shared rAF loop to `index.html`**, near the top of the JS section (before `initRingTilt` and `initTerrainDrag`):

  ```js
  let _animFrameRunning = false;

  function startAnimFrame() {
    if (_animFrameRunning) return;
    _animFrameRunning = true;
    requestAnimationFrame(animFrame);
  }

  function animFrame() {
    let stillRunning = false;

    // --- Ring tilt spring-back ---
    const tilt = window._frTilt;
    if (tilt && !tilt.dragging) {
      if (Math.abs(tilt.velX) > 0.3 || Math.abs(tilt.velY) > 0.3) {
        tilt.velX *= 0.88;
        tilt.velY *= 0.88;
        tilt.rotX *= 0.88;
        tilt.rotY *= 0.88;
        tilt.applyTransform();
        stillRunning = true;
      } else if (Math.abs(tilt.rotX) > 0.1 || Math.abs(tilt.rotY) > 0.1) {
        // Snap to flat
        tilt.rotX = 0; tilt.rotY = 0;
        tilt.velX = 0; tilt.velY = 0;
        const tiltEl = tilt.tilt;
        if (tiltEl) {
          tiltEl.style.transition = 'transform 0.15s ease-out';
          tiltEl.style.transform = 'rotateX(0deg) rotateY(0deg)';
          setTimeout(() => { if (tiltEl) tiltEl.style.transition = ''; }, 200);
        }
      }
    }

    // --- Terrain momentum ---
    const terrain = window._frTerrain;
    if (terrain && !terrain.isDragging && terrain.velocity && Math.abs(terrain.velocity) > 0.5) {
      terrain.velocity *= 0.88;
      const inner = document.getElementById('fr-terrain-inner');
      const stage = document.getElementById('fr-terrain-stage');
      if (inner && stage) {
        const totalW = terrain.totalW || inner.scrollWidth;
        const viewportW = stage.offsetWidth || 320;
        const newOffset = Math.max(-(totalW - viewportW), Math.min(0, (terrain.offset || 0) + terrain.velocity));
        terrain.offset = newOffset;
        inner.style.transform = `translateX(${newOffset}px)`;
      }
      stillRunning = true;
    }

    if (stillRunning) {
      requestAnimationFrame(animFrame);
    } else {
      _animFrameRunning = false;
    }
  }
  ```

## Key Pitfalls to Avoid

1. **`box-shadow` inside `rotateX` / `preserve-3d` on iOS Safari** — always use `filter: drop-shadow()` for the active terrain bar glow. `box-shadow` does not render correctly in 3D transforms on iOS.

2. **SVG `<path>` elements are not natively interactive** — both `role="button"` AND `tabindex="0"` are required for VoiceOver/iOS accessibility. Missing either breaks screen reader access.

3. **`monthList` must be populated before `sw()` resets it** — `renderForgeRing()` calls `monthList = getMonthList()` at the top. `sw()` only resets `currentMonthIdx` (not `monthList`), so the array persists across tab switches correctly.

4. **`dayPct()` returns `null` when no habits exist for a day** — always guard with `if (p !== null)` before using the value. Early months before habits were configured should not drag the average to 0.

5. **`transform: translateX` not `scrollLeft`** — `fr-terrain-inner` is moved exclusively via CSS transform. Never use `scrollLeft` on this element or layout reflows will occur and 60fps will be lost.

6. **Single `animFrame()` loop** — both ring tilt and terrain momentum feed into the same loop via `window._frTilt` and `window._frTerrain`. Never start separate `requestAnimationFrame` calls for each — this doubles rAF overhead on older iPhones.

7. **`filter: drop-shadow` in SVG `<style>` tags** — the `todayWedge` keyframe is defined inside the SVG's own `<style>` tag (not in the external CSS file) because SVG `filter` animations scoped this way are more reliable on Capacitor/WKWebView.
